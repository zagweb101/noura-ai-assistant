   393→  const playTTS = async (text: string): Promise<void> => {
   394→    return new Promise(async (resolve) => {
   395→      try {
   396→        // Priority 1: Use TTS API if key is saved
   397→        const savedKey = localStorage.getItem('tts_api_key')
   398→        const savedProvider = localStorage.getItem('tts_provider') || 'auto'
   399→        const savedVoice = localStorage.getItem('tts_voice') || 'nova'
   400→        if (savedKey) {
   401→          try {
   402→            const res = await fetch('/api/tts', {
   403→              method: 'POST',
   404→              headers: { 'Content-Type': 'application/json' },
   405→              body: JSON.stringify({ text, speed: 1.0, api_key: savedKey, provider: savedProvider, voice: savedVoice }),
   406→            })
   407→
   408→            if (res.ok) {
   409→              const audioBlob = await res.blob()
   410→              const audioUrl = URL.createObjectURL(audioBlob)
   411→              const audio = new Audio(audioUrl)
   412→              audioRef.current = audio
   413→
   414→              audio.onended = () => {
   415→                URL.revokeObjectURL(audioUrl)
   416→                audioRef.current = null
   417→                resolve()
   418→              }
   419→              audio.onerror = () => {
   420→                URL.revokeObjectURL(audioUrl)
   421→                audioRef.current = null
   422→                // Fall back to browser TTS
   423→                browserTTS(text, resolve)
   424→              }
   425→
   426→              await audio.play()
   427→              return // Success, don't fall through
   428→            }
   429→          } catch (error) {
   430→            console.error('API TTS failed, falling back to browser:', error)
   431→          }
   432→        }
   433→
   434→        // Priority 2: Use browser's built-in SpeechSynthesis as fallback
   435→        browserTTS(text, resolve)
   436→      } catch (error) {
   437→        console.error('TTS playback error:', error)
   438→        resolve()
   439→      }
   440→    })
   441→  }

   443→  const browserTTS = (text: string, resolve: (value: void) => void) => {
   444→    if ('speechSynthesis' in window) {
   445→      window.speechSynthesis.cancel()
   446→
   447→      const utterance = new SpeechSynthesisUtterance(text)
   448→      utterance.lang = 'ar-SA'
   449→      utterance.rate = 0.95
   450→      utterance.pitch = 1.0
   451→      utterance.volume = 1.0
   452→
   453→      const voices = window.speechSynthesis.getVoices()
   454→      const arabicVoice = voices.find(v => v.lang.startsWith('ar'))
   455→      if (arabicVoice) {
   456→        utterance.voice = arabicVoice
   457→      }
   458→
   459→      utterance.onend = () => resolve()
   460→      utterance.onerror = () => resolve()
   461→
   462→      if (voices.length === 0) {
   463→        window.speechSynthesis.onvoiceschanged = () => {
   464→          const newVoices = window.speechSynthesis.getVoices()
   465→          const arVoice = newVoices.find(v => v.lang.startsWith('ar'))
   466→          if (arVoice) utterance.voice = arVoice
   467→          window.speechSynthesis.speak(utterance)
   468→        }
   469→      } else {
   470→        window.speechSynthesis.speak(utterance)
   471→      }
   472→    } else {
   473→      console.warn('SpeechSynthesis not supported')
   474→      resolve()
   475→    }
