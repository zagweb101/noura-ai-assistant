   394→  const playTTS = async (text: string): Promise<void> => {
   395→    return new Promise(async (resolve) => {
   396→      try {
   397→        // Priority 1: Use TTS API if key is saved
   398→        const savedKey = localStorage.getItem('tts_api_key')
   399→        const savedProvider = localStorage.getItem('tts_provider') || 'auto'
   400→        const savedVoice = localStorage.getItem('tts_voice') || 'alloy'
   401→        if (savedKey) {
   402→          try {
   403→            const res = await fetch('/api/tts', {
   404→              method: 'POST',
   405→              headers: { 'Content-Type': 'application/json' },
   406→              body: JSON.stringify({ text, speed: 1.0, api_key: savedKey, provider: savedProvider, voice: savedVoice }),
   407→            })
   408→
   409→            if (res.ok) {
   410→              const audioBlob = await res.blob()
   411→              const audioUrl = URL.createObjectURL(audioBlob)
   412→              const audio = new Audio(audioUrl)
   413→              audioRef.current = audio
   414→
   415→              audio.onended = () => {
   416→                URL.revokeObjectURL(audioUrl)
   417→                audioRef.current = null
   418→                resolve()
   419→              }
   420→              audio.onerror = () => {
   421→                URL.revokeObjectURL(audioUrl)
   422→                audioRef.current = null
   423→                // Fall back to browser TTS
   424→                browserTTS(text, resolve)
   425→              }
   426→
   427→              await audio.play()
   428→              return // Success, don't fall through
   429→            }
   430→          } catch (error) {
   431→            console.error('API TTS failed, falling back to browser:', error)
   432→          }
   433→        }
   434→
   435→        // Priority 2: Use browser's built-in SpeechSynthesis as fallback
   436→        browserTTS(text, resolve)
   437→      } catch (error) {
   438→        console.error('TTS playback error:', error)
   439→        resolve()
   440→      }
   441→    })
   442→  }

   444→  const browserTTS = (text: string, resolve: (value: void) => void) => {
   445→    if ('speechSynthesis' in window) {
   446→      window.speechSynthesis.cancel()
   447→
   448→      const utterance = new SpeechSynthesisUtterance(text)
   449→      utterance.lang = 'ar-SA'
   450→      utterance.rate = 0.95
   451→      utterance.pitch = 1.0
   452→      utterance.volume = 1.0
   453→
   454→      const voices = window.speechSynthesis.getVoices()
   455→      const arabicVoice = voices.find(v => v.lang.startsWith('ar'))
   456→      if (arabicVoice) {
   457→        utterance.voice = arabicVoice
   458→      }
   459→
   460→      utterance.onend = () => resolve()
   461→      utterance.onerror = () => resolve()
   462→
   463→      if (voices.length === 0) {
   464→        window.speechSynthesis.onvoiceschanged = () => {
   465→          const newVoices = window.speechSynthesis.getVoices()
   466→          const arVoice = newVoices.find(v => v.lang.startsWith('ar'))
   467→          if (arVoice) utterance.voice = arVoice
   468→          window.speechSynthesis.speak(utterance)
   469→        }
   470→      } else {
   471→        window.speechSynthesis.speak(utterance)
   472→      }
   473→    } else {
   474→      console.warn('SpeechSynthesis not supported')
   475→      resolve()
   476→    }
